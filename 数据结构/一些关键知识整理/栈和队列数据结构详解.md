## 栈的基本概念

### 什么是栈？

栈（Stack）是一种**后进先出**（Last In First Out, LIFO）的线性数据结构。栈具有以下特点：

1. **LIFO原则**：最后进入的元素最先被取出
2. **限制性访问**：只能在栈顶进行插入和删除操作
3. **线性结构**：元素之间存在一对一的关系
4. **有限容量**：栈的容量是有限的

### 栈的基本操作

| 操作名称 | 功能描述 | 输入 | 输出 |
|----------|----------|------|------|
| **Push** | 入栈，在栈顶插入元素 | 栈，元素 | 更新后的栈 |
| **Pop** | 出栈，删除栈顶元素 | 栈 | 被删除的元素 |
| **Top/Peek** | 查看栈顶元素 | 栈 | 栈顶元素值 |
| **IsEmpty** | 判断栈是否为空 | 栈 | 布尔值 |
| **Size** | 获取栈中元素个数 | 栈 | 整数 |

### 栈的抽象数据类型

```cpp
ADT Stack {
    数据对象：D = {a1, a2, ..., an}
    数据关系：R = {<ai-1, ai> | ai-1, ai ∈ D, i = 2, 3, ..., n}
    
    基本操作：
    InitStack(&S)        // 初始化栈
    DestroyStack(&S)     // 销毁栈
    ClearStack(&S)       // 清空栈
    StackEmpty(S)        // 判空
    StackLength(S)       // 求长度
    GetTop(S, &e)        // 获取栈顶元素
    Push(&S, e)          // 入栈
    Pop(&S, &e)          // 出栈
} ADT Stack
```

### 栈的示意图

```
栈顶 ←→ 元素3
      ↓
      元素2
      ↓
      元素1
      ↓
栈底 ←→ 元素0
```

## 栈的实现

### 1. 顺序栈（数组实现）

```cpp
#include <iostream>
#include <stdexcept>
using namespace std;

template<typename T>
class ArrayStack {
private:
    T* data;           // 数据数组
    int capacity;      // 容量
    int top;           // 栈顶指针
    
public:
    // 构造函数
    ArrayStack(int cap = 100) : capacity(cap), top(-1) {
        data = new T[capacity];
    }
    
    // 析构函数
    ~ArrayStack() {
        delete[] data;
    }
    
    // 拷贝构造函数
    ArrayStack(const ArrayStack& other) {
        capacity = other.capacity;
        top = other.top;
        data = new T[capacity];
        for (int i = 0; i <= top; i++) {
            data[i] = other.data[i];
        }
    }
    
    // 赋值操作符
    ArrayStack& operator=(const ArrayStack& other) {
        if (this != &other) {
            delete[] data;
            capacity = other.capacity;
            top = other.top;
            data = new T[capacity];
            for (int i = 0; i <= top; i++) {
                data[i] = other.data[i];
            }
        }
        return *this;
    }
    
    // 判断是否为空
    bool isEmpty() const {
        return top == -1;
    }
    
    // 判断是否已满
    bool isFull() const {
        return top == capacity - 1;
    }
    
    // 获取栈中元素个数
    int size() const {
        return top + 1;
    }
    
    // 获取容量
    int getCapacity() const {
        return capacity;
    }
    
    // 获取栈顶元素
    T peek() const {
        if (isEmpty()) {
            throw runtime_error("栈为空");
        }
        return data[top];
    }
    
    // 入栈
    void push(const T& value) {
        if (isFull()) {
            expand();  // 扩容
        }
        data[++top] = value;
    }
    
    // 出栈
    T pop() {
        if (isEmpty()) {
            throw runtime_error("栈为空");
        }
        return data[top--];
    }
    
    // 清空栈
    void clear() {
        top = -1;
    }
    
    // 打印栈
    void print() const {
        cout << "栈顶 -> ";
        for (int i = top; i >= 0; i--) {
            cout << data[i];
            if (i > 0) cout << " -> ";
        }
        cout << " -> 栈底" << endl;
    }
    
private:
    // 扩容
    void expand() {
        int newCapacity = capacity * 2;
        T* newData = new T[newCapacity];
        
        for (int i = 0; i <= top; i++) {
            newData[i] = data[i];
        }
        
        delete[] data;
        data = newData;
        capacity = newCapacity;
    }
};
```

### 2. 链式栈（链表实现）

```cpp
template<typename T>
class LinkedStack {
private:
    struct Node {
        T data;
        Node* next;
        
        Node(const T& value) : data(value), next(nullptr) {}
    };
    
    Node* top;          // 栈顶指针
    int length;         // 栈中元素个数
    
public:
    // 构造函数
    LinkedStack() : top(nullptr), length(0) {}
    
    // 析构函数
    ~LinkedStack() {
        clear();
    }
    
    // 拷贝构造函数
    LinkedStack(const LinkedStack& other) {
        top = nullptr;
        length = 0;
        
        // 从栈底到栈顶复制元素
        LinkedStack temp;
        Node* current = other.top;
        while (current != nullptr) {
            temp.push(current->data);
            current = current->next;
        }
        
        // 将临时栈的元素转移到当前栈
        while (!temp.isEmpty()) {
            push(temp.pop());
        }
    }
    
    // 赋值操作符
    LinkedStack& operator=(const LinkedStack& other) {
        if (this != &other) {
            clear();
            
            // 从栈底到栈顶复制元素
            LinkedStack temp;
            Node* current = other.top;
            while (current != nullptr) {
                temp.push(current->data);
                current = current->next;
            }
            
            // 将临时栈的元素转移到当前栈
            while (!temp.isEmpty()) {
                push(temp.pop());
            }
        }
        return *this;
    }
    
    // 判断是否为空
    bool isEmpty() const {
        return top == nullptr;
    }
    
    // 获取栈中元素个数
    int size() const {
        return length;
    }
    
    // 获取栈顶元素
    T peek() const {
        if (isEmpty()) {
            throw runtime_error("栈为空");
        }
        return top->data;
    }
    
    // 入栈
    void push(const T& value) {
        Node* newNode = new Node(value);
        newNode->next = top;
        top = newNode;
        length++;
    }
    
    // 出栈
    T pop() {
        if (isEmpty()) {
            throw runtime_error("栈为空");
        }
        
        Node* toDelete = top;
        T value = top->data;
        top = top->next;
        delete toDelete;
        length--;
        return value;
    }
    
    // 清空栈
    void clear() {
        while (top != nullptr) {
            Node* toDelete = top;
            top = top->next;
            delete toDelete;
        }
        length = 0;
    }
    
    // 打印栈
    void print() const {
        cout << "栈顶 -> ";
        Node* current = top;
        while (current != nullptr) {
            cout << current->data;
            if (current->next != nullptr) cout << " -> ";
            current = current->next;
        }
        cout << " -> 栈底" << endl;
    }
};
```

### 1. 表达式求值

**中缀表达式转后缀表达式**：

```cpp
class ExpressionEvaluator {
private:
    ArrayStack<char> operatorStack;
    ArrayStack<int> operandStack;
    
    // 获取运算符优先级
    int getPriority(char op) {
        switch (op) {
            case '+': case '-': return 1;
            case '*': case '/': return 2;
            case '^': return 3;
            default: return 0;
        }
    }
    
    // 判断是否为运算符
    bool isOperator(char c) {
        return c == '+' || c == '-' || c == '*' || c == '/' || c == '^';
    }
    
    // 判断是否为数字
    bool isDigit(char c) {
        return c >= '0' && c <= '9';
    }
    
public:
    // 中缀表达式转后缀表达式
    string infixToPostfix(const string& infix) {
        string postfix = "";
        operatorStack.clear();
        
        for (char c : infix) {
            if (isDigit(c)) {
                postfix += c;
            } else if (c == '(') {
                operatorStack.push(c);
            } else if (c == ')') {
                while (!operatorStack.isEmpty() && operatorStack.peek() != '(') {
                    postfix += operatorStack.pop();
                }
                if (!operatorStack.isEmpty()) {
                    operatorStack.pop();  // 弹出 '('
                }
            } else if (isOperator(c)) {
                while (!operatorStack.isEmpty() && 
                       operatorStack.peek() != '(' &&
                       getPriority(operatorStack.peek()) >= getPriority(c)) {
                    postfix += operatorStack.pop();
                }
                operatorStack.push(c);
            }
        }
        
        while (!operatorStack.isEmpty()) {
            postfix += operatorStack.pop();
        }
        
        return postfix;
    }
    
    // 计算后缀表达式
    int evaluatePostfix(const string& postfix) {
        operandStack.clear();
        
        for (char c : postfix) {
            if (isDigit(c)) {
                operandStack.push(c - '0');
            } else if (isOperator(c)) {
                int b = operandStack.pop();
                int a = operandStack.pop();
                int result = 0;
                
                switch (c) {
                    case '+': result = a + b; break;
                    case '-': result = a - b; break;
                    case '*': result = a * b; break;
                    case '/': result = a / b; break;
                    case '^': result = pow(a, b); break;
                }
                
                operandStack.push(result);
            }
        }
        
        return operandStack.pop();
    }
};
```

### 2. 括号匹配

```cpp
class BracketMatcher {
public:
    bool isValid(const string& s) {
        ArrayStack<char> stack;
        
        for (char c : s) {
            if (c == '(' || c == '[' || c == '{') {
                stack.push(c);
            } else if (c == ')' || c == ']' || c == '}') {
                if (stack.isEmpty()) {
                    return false;
                }
                
                char top = stack.pop();
                if ((c == ')' && top != '(') ||
                    (c == ']' && top != '[') ||
                    (c == '}' && top != '{')) {
                    return false;
                }
            }
        }
        
        return stack.isEmpty();
    }
};
```

## 队列的基本概念

### 什么是队列？

队列（Queue）是一种**先进先出**（First In First Out, FIFO）的线性数据结构。队列具有以下特点：

1. **FIFO原则**：最先进入的元素最先被取出
2. **限制性访问**：只能在队尾插入，在队头删除
3. **线性结构**：元素之间存在一对一的关系
4. **有限容量**：队列的容量是有限的

### 队列的基本操作

| 操作名称 | 功能描述 | 输入 | 输出 |
|----------|----------|------|------|
| **Enqueue** | 入队，在队尾插入元素 | 队列，元素 | 更新后的队列 |
| **Dequeue** | 出队，删除队头元素 | 队列 | 被删除的元素 |
| **Front** | 查看队头元素 | 队列 | 队头元素值 |
| **Rear** | 查看队尾元素 | 队列 | 队尾元素值 |
| **IsEmpty** | 判断队列是否为空 | 队列 | 布尔值 |
| **Size** | 获取队列中元素个数 | 队列 | 整数 |

### 队列的抽象数据类型

```cpp
ADT Queue {
    数据对象：D = {a1, a2, ..., an}
    数据关系：R = {<ai-1, ai> | ai-1, ai ∈ D, i = 2, 3, ..., n}
    
    基本操作：
    InitQueue(&Q)        // 初始化队列
    DestroyQueue(&Q)     // 销毁队列
    ClearQueue(&Q)       // 清空队列
    QueueEmpty(Q)        // 判空
    QueueLength(Q)       // 求长度
    GetHead(Q, &e)       // 获取队头元素

    DeQueue(&Q, &e)      // 出队
} ADT Queue
```

### 队列的示意图

```

```

```cpp
template<typename T>
class ArrayQueue {
private:
    T* data;           // 数据数组
    int capacity;      // 容量
    int front;         // 队头指针
    int rear;          // 队尾指针
    int length;        // 队列长度
    
public:
    // 构造函数
    ArrayQueue(int cap = 100) : capacity(cap), front(0), rear(0), length(0) {
        data = new T[capacity];
    }
    
    // 析构函数
    ~ArrayQueue() {
        delete[] data;
    }
    
    // 拷贝构造函数
    ArrayQueue(const ArrayQueue& other) {
        capacity = other.capacity;
        front = other.front;
        rear = other.rear;
        length = other.length;
        data = new T[capacity];
        
        for (int i = 0; i < capacity; i++) {
            data[i] = other.data[i];
        }
    }
    
    // 赋值操作符
    ArrayQueue& operator=(const ArrayQueue& other) {
        if (this != &other) {
            delete[] data;
            capacity = other.capacity;
            front = other.front;
            rear = other.rear;
            length = other.length;
            data = new T[capacity];
            
            for (int i = 0; i < capacity; i++) {
                data[i] = other.data[i];
            }
        }
        return *this;
    }
    
    // 判断是否为空
    bool isEmpty() const {
        return length == 0;
    }
    
    // 判断是否已满
    bool isFull() const {
        return length == capacity;
    }
    
    // 获取队列中元素个数
    int size() const {
        return length;
    }
    
    // 获取容量
    int getCapacity() const {
        return capacity;
    }
    
    // 获取队头元素
    T front() const {
        if (isEmpty()) {
            throw runtime_error("队列为空");
        }
        return data[front];
    }
    
    // 获取队尾元素
    T rear() const {
        if (isEmpty()) {
            throw runtime_error("队列为空");
        }
        return data[(rear - 1 + capacity) % capacity];
    }
    
    // 入队
    void enqueue(const T& value) {
        if (isFull()) {
            expand();  // 扩容
        }
        
        data[rear] = value;
        rear = (rear + 1) % capacity;
        length++;
    }
    
    // 出队
    T dequeue() {
        if (isEmpty()) {
            throw runtime_error("队列为空");
        }
        
        T value = data[front];
        front = (front + 1) % capacity;
        length--;
        return value;
    }
    
    // 清空队列
    void clear() {
        front = 0;
        rear = 0;
        length = 0;
    }
    
    // 打印队列
    void print() const {
        cout << "队头 -> ";
        for (int i = 0; i < length; i++) {
            int index = (front + i) % capacity;
            cout << data[index];
            if (i < length - 1) cout << " -> ";
        }
        cout << " -> 队尾" << endl;
    }
    
private:
    // 扩容
    void expand() {
        int newCapacity = capacity * 2;
        T* newData = new T[newCapacity];
        
        for (int i = 0; i < length; i++) {
            int index = (front + i) % capacity;
            newData[i] = data[index];
        }
        
        delete[] data;
        data = newData;
        front = 0;
        rear = length;
        capacity = newCapacity;
    }
};
```

### 2. 链式队列（链表实现）

```cpp
template<typename T>
class LinkedQueue {
private:
    struct Node {
        T data;
        Node* next;
        
        Node(const T& value) : data(value), next(nullptr) {}
    };
    
    Node* front;        // 队头指针
    Node* rear;         // 队尾指针
    int length;         // 队列长度
    
public:
    // 构造函数
    LinkedQueue() : front(nullptr), rear(nullptr), length(0) {}
    
    // 析构函数
    ~LinkedQueue() {
        clear();
    }
    
    // 拷贝构造函数
    LinkedQueue(const LinkedQueue& other) {
        front = nullptr;
        rear = nullptr;
        length = 0;
        
        Node* current = other.front;
        while (current != nullptr) {
            enqueue(current->data);
            current = current->next;
        }
    }
    
    // 赋值操作符
    LinkedQueue& operator=(const LinkedQueue& other) {
        if (this != &other) {
            clear();
            
            Node* current = other.front;
            while (current != nullptr) {
                enqueue(current->data);
                current = current->next;
            }
        }
        return *this;
    }
    
    // 判断是否为空
    bool isEmpty() const {
        return front == nullptr;
    }
    
    // 获取队列中元素个数
    int size() const {
        return length;
    }
    
    // 获取队头元素
    T front() const {
        if (isEmpty()) {
            throw runtime_error("队列为空");
        }
        return front->data;
    }
    
    // 获取队尾元素
    T rear() const {
        if (isEmpty()) {
            throw runtime_error("队列为空");
        }
        return rear->data;
    }
    
    // 入队
    void enqueue(const T& value) {
        Node* newNode = new Node(value);
        
        if (isEmpty()) {
            front = rear = newNode;
        } else {
            rear->next = newNode;
            rear = newNode;
        }
        
        length++;
    }
    
    // 出队
    T dequeue() {
        if (isEmpty()) {
            throw runtime_error("队列为空");
        }
        
        Node* toDelete = front;
        T value = front->data;
        front = front->next;
        
        if (front == nullptr) {
            rear = nullptr;
        }
        
        delete toDelete;
        length--;
        return value;
    }
    
    // 清空队列
    void clear() {
        while (front != nullptr) {
            Node* toDelete = front;
            front = front->next;
            delete toDelete;
        }
        rear = nullptr;
        length = 0;
    }
    
    // 打印队列
    void print() const {
        cout << "队头 -> ";
        Node* current = front;
        while (current != nullptr) {
            cout << current->data;
            if (current->next != nullptr) cout << " -> ";
            current = current->next;
        }
        cout << " -> 队尾" << endl;
    }
};
```

```cpp
class Graph {

    vector<vector<int>> adjList;

    Graph(int v) : vertices(v) {
        adjList.resize(v);
    }

            cout << vertex << " ";
            
            for (int neighbor : adjList[vertex]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    queue.enqueue(neighbor);
                }
            }
        }
        cout << endl;
    }
};
```

## 双端队列

### 双端队列的定义

双端队列（Deque，Double-ended Queue）是一种允许在两端进行插入和删除操作的线性数据结构。

### 双端队列的特点

1. **两端操作**：可以在队头和队尾进行插入和删除
2. **灵活访问**：支持随机访问和顺序访问
3. **高效操作**：两端操作的时间复杂度都是O(1)

### 双端队列实现

```cpp
template<typename T>
class Deque {
private:
    struct Node {
        T data;
        Node* prev;
        Node* next;
        
        Node(const T& value) : data(value), prev(nullptr), next(nullptr) {}
    };
    
    Node* front;
    Node* rear;
    int length;
    
public:
    Deque() : front(nullptr), rear(nullptr), length(0) {}
    
    ~Deque() {
        clear();
    }
    
    bool isEmpty() const {
        return front == nullptr;
    }
    
    int size() const {
        return length;
    }
    
    // 在队头插入
    void pushFront(const T& value) {
        Node* newNode = new Node(value);
        
        if (isEmpty()) {
            front = rear = newNode;
        } else {
            newNode->next = front;
            front->prev = newNode;
            front = newNode;
        }
        
        length++;
    }
    
    // 在队尾插入
    void pushBack(const T& value) {
        Node* newNode = new Node(value);
        
        if (isEmpty()) {
            front = rear = newNode;
        } else {
            rear->next = newNode;
            newNode->prev = rear;
            rear = newNode;
        }
        
        length++;
    }
    
    // 在队头删除
    T popFront() {
        if (isEmpty()) {
            throw runtime_error("双端队列为空");
        }
        
        Node* toDelete = front;
        T value = front->data;
        front = front->next;
        
        if (front == nullptr) {
            rear = nullptr;
        } else {
            front->prev = nullptr;
        }
        
        delete toDelete;
        length--;
        return value;
    }
    
    // 在队尾删除
    T popBack() {
        if (isEmpty()) {
            throw runtime_error("双端队列为空");
        }
        
        Node* toDelete = rear;
        T value = rear->data;
        rear = rear->prev;
        
        if (rear == nullptr) {
            front = nullptr;
        } else {
            rear->next = nullptr;
        }
        
        delete toDelete;
        length--;
        return value;
    }
    
    // 获取队头元素
    T getFront() const {
        if (isEmpty()) {
            throw runtime_error("双端队列为空");
        }
        return front->data;
    }
    
    // 获取队尾元素
    T getBack() const {
        if (isEmpty()) {
            throw runtime_error("双端队列为空");
        }
        return rear->data;
    }
    
    void clear() {
        while (front != nullptr) {
            Node* toDelete = front;
            front = front->next;
            delete toDelete;
        }
        rear = nullptr;
        length = 0;
    }
    
    void print() const {
        cout << "队头 -> ";
        Node* current = front;
        while (current != nullptr) {
            cout << current->data;
            if (current->next != nullptr) cout << " -> ";
            current = current->next;
        }
        cout << " -> 队尾" << endl;
    }
};
```

## 优先队列

### 优先队列的定义

优先队列（Priority Queue）是一种特殊的队列，其中每个元素都有一个优先级，优先级高的元素先被处理。

### 优先队列的特点

1. **优先级排序**：元素按优先级排序
2. **动态调整**：优先级可以动态改变
3. **高效操作**：插入和删除操作的时间复杂度为O(log n)

### 优先队列实现

```cpp
template<typename T>
class PriorityQueue {
private:
    vector<T> heap;
    function<bool(const T&, const T&)> compare;
    
    void heapifyUp(int index) {
        while (index > 0) {
            int parent = (index - 1) / 2;
            if (compare(heap[index], heap[parent])) {
                swap(heap[index], heap[parent]);
                index = parent;
            } else {
                break;
            }
        }
    }
    
    void heapifyDown(int index) {
        int left = 2 * index + 1;
        int right = 2 * index + 2;
        int largest = index;
        
        if (left < heap.size() && compare(heap[left], heap[largest])) {
            largest = left;
        }
        
        if (right < heap.size() && compare(heap[right], heap[largest])) {
            largest = right;
        }
        
        if (largest != index) {
            swap(heap[index], heap[largest]);
            heapifyDown(largest);
        }
    }
    
public:
    // 构造函数，默认最大堆
    PriorityQueue(function<bool(const T&, const T&)> comp = [](const T& a, const T& b) { return a > b; })
        : compare(comp) {}
    
    bool isEmpty() const {
        return heap.empty();
    }
    
    int size() const {
        return heap.size();
    }
    
    void push(const T& value) {
        heap.push_back(value);
        heapifyUp(heap.size() - 1);
    }
    
    T top() const {
        if (isEmpty()) {
            throw runtime_error("优先队列为空");
        }
        return heap[0];
    }
    
    void pop() {
        if (isEmpty()) {
            throw runtime_error("优先队列为空");
        }
        
        heap[0] = heap.back();
        heap.pop_back();
        
        if (!heap.empty()) {
            heapifyDown(0);
        }
    }
    
    void print() const {
        cout << "优先队列: ";
        for (const T& item : heap) {
            cout << item << " ";
        }
        cout << endl;
    }
};
```

## 算法复杂度分析

### 时间复杂度对比

| 操作 | 顺序栈 | 链式栈 | 顺序队列 | 链式队列 | 双端队列 | 优先队列 |
|------|--------|--------|----------|----------|----------|----------|
| **插入** | O(1) | O(1) | O(1) | O(1) | O(1) | O(log n) |
| **删除** | O(1) | O(1) | O(1) | O(1) | O(1) | O(log n) |
| **查看顶部** | O(1) | O(1) | O(1) | O(1) | O(1) | O(1) |
| **查找** | O(n) | O(n) | O(n) | O(n) | O(n) | O(n) |

### 空间复杂度对比

| 数据结构 | 空间复杂度 | 说明 |
|----------|------------|------|
| **顺序栈** | O(n) | 连续存储，无额外指针 |
| **链式栈** | O(n) | 每个节点一个指针 |
| **顺序队列** | O(n) | 连续存储，无额外指针 |
| **链式队列** | O(n) | 每个节点一个指针 |
| **双端队列** | O(n) | 每个节点两个指针 |
| **优先队列** | O(n) | 堆存储 |

### 性能特点总结

#### 栈的特点
1. **LIFO原则**：后进先出，适合撤销操作
2. **递归实现**：函数调用栈的天然实现
3. **表达式求值**：中缀转后缀，后缀表达式求值

#### 队列的特点
1. **FIFO原则**：先进先出，适合任务调度
2. **广度优先搜索**：BFS算法的核心数据结构
3. **缓冲区管理**：数据流控制

#### 双端队列的特点
1. **灵活操作**：两端都可以插入删除
2. **滑动窗口**：适合滑动窗口算法
3. **单调队列**：维护单调性

#### 优先队列的特点
1. **优先级排序**：按优先级处理元素
2. **堆排序**：基于堆的排序算法
3. **图算法**：Dijkstra算法、Prim算法

## 实际应用案例

### 案例1：计算器实现

```cpp
class Calculator {
private:
    ArrayStack<double> operandStack;
    ArrayStack<char> operatorStack;
    
    int getPriority(char op) {
        switch (op) {
            case '+': case '-': return 1;
            case '*': case '/': return 2;
            case '^': return 3;
            default: return 0;
        }
    }
    
    double calculate(double a, double b, char op) {
        switch (op) {
            case '+': return a + b;
            case '-': return a - b;
            case '*': return a * b;
            case '/': return a / b;
            case '^': return pow(a, b);
            default: return 0;
        }
    }
    
public:
    double evaluate(const string& expression) {
        operandStack.clear();
        operatorStack.clear();
        
        for (char c : expression) {
            if (isdigit(c)) {
                operandStack.push(c - '0');
            } else if (c == '(') {
                operatorStack.push(c);
            } else if (c == ')') {
                while (!operatorStack.isEmpty() && operatorStack.peek() != '(') {
                    char op = operatorStack.pop();
                    double b = operandStack.pop();
                    double a = operandStack.pop();
                    operandStack.push(calculate(a, b, op));
                }
                operatorStack.pop();  // 弹出 '('
            } else if (c == '+' || c == '-' || c == '*' || c == '/' || c == '^') {
                while (!operatorStack.isEmpty() && 
                       operatorStack.peek() != '(' &&
                       getPriority(operatorStack.peek()) >= getPriority(c)) {
                    char op = operatorStack.pop();
                    double b = operandStack.pop();
                    double a = operandStack.pop();
                    operandStack.push(calculate(a, b, op));
                }
                operatorStack.push(c);
            }
        }
        
        while (!operatorStack.isEmpty()) {
            char op = operatorStack.pop();
            double b = operandStack.pop();
            double a = operandStack.pop();
            operandStack.push(calculate(a, b, op));
        }
        
        return operandStack.pop();
    }
};
```

```cpp
class Process {

    PriorityQueue<Process> readyQueue;

        Process process = readyQueue.top();
        readyQueue.pop();
        cout << "调度进程: " << process << endl;
        return process;

        return !readyQueue.isEmpty();
    }

    }
```

```cpp
class BrowserHistory {
private:
    ArrayStack<string> backStack;
    ArrayStack<string> forwardStack;
    string currentPage;

    void visit(const string& url) {
        if (!currentPage.empty()) {

        backStack.push(currentPage);
        currentPage = forwardStack.pop();
        cout << "前进到: " << currentPage << endl;
        return currentPage;
    }

        return currentPage;
    }

        cout << "当前页面: " << currentPage << endl;
        cout << "后退栈: ";
        backStack.print();
        cout << "前进栈: ";
```

## 总结

### 栈和队列的重要性

1. **基础数据结构**：是其他复杂数据结构的基础
2. **算法实现**：许多算法都基于栈和队列实现
3. **系统设计**：在操作系统、编译器等领域有重要应用

### 选择原则

#### 选择栈的情况
- 需要撤销操作（如编辑器、浏览器）
- 表达式求值
- 递归算法实现
- 深度优先搜索

#### 选择队列的情况
- 任务调度
- 广度优先搜索
- 缓冲区管理
- 生产者消费者模式

#### 选择双端队列的情况
- 滑动窗口算法
- 单调队列
- 需要两端操作

#### 选择优先队列的情况
- 任务优先级调度
- 图算法（Dijkstra、Prim）
- 堆排序
- 事件驱动模拟

1. **理解基本概念**：LIFO、FIFO、优先级
